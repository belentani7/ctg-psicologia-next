```python
def speech_to_text(audio_file_path):

            file=audio_file,

        )

def text_to_speech(text):
    """Genera audio con voz humana moderna usando ElevenLabs."""
    url = f"https://api.elevenlabs.io/v1/text-to-speech/{ELEVENLABS_VOICE_ID}"

            "similarity_boost": 0.8,
            "style": 0.4, # Toque conversacional
            "use_speaker_boost": True
```

```python
import random
import time

class MateEscape:
    def __init__(self):
        self.puntos = 0
        self.vidas = 3
        self.glosario = {
            "ecuación": "equação",
            "despejar": "isolar",
            "solución": "solução",
            "incógnita": "incógnita",
            "sumar": "somar",
            "restar": "subtrair",
            "multiplicar": "multiplicar",
            "dividir": "dividir"
        }
    
    def generar_ecuacion(self, nivel):
        if nivel == 1:
            # Tipo: x + a = b
            x = random.randint(1, 10)
            a = random.randint(1, 9)
            b = x + a
            return f"x + {a} = {b}", x
        
        elif nivel == 2:
            # Tipo: ax + b = c
            x = random.randint(1, 8)
            a = random.randint(2, 5)
            b = random.randint(1, 10)
            c = a * x + b
            return f"{a}x + {b} = {c}", x
        
        else:
            # Tipo: ax - b = c
            x = random.randint(2, 10)
            a = random.randint(2, 6)
            b = random.randint(1, 8)
            c = a * x - b
            return f"{a}x - {b} = {c}", x
    
    def jugar(self):
        print("\n " + "="*50)
        print("🧮 JUEGO: MATE-ESCAPE")
        print("="*50)
        print("Resuelve ecuaciones antes de que se agote el tiempo.")
        print("Tienes 3 vidas. ¡Llega al nivel 3!\n")

        for es, pt in self.glosario.items():
            print(f"  {es.upper()} = {pt}")
        print("="*50)
        time.sleep(2)
        
        nivel = 1
        while self.vidas > 0 and nivel <= 3:
            print(f"\n📊 NIVEL {nivel} | Vidas: {'❤️' * self.vidas}")
            
            ecuacion, solucion = self.generar_ecuacion(nivel)
            print(f"\nResuelve: **{ecuacion}**")
            print(f"¿Cuánto vale x?")
            
            inicio = time.time()
            try:
                respuesta = int(input("x = "))
                tiempo = time.time() - inicio
                
                if tiempo > 15:
                    print(" ¡TIEMPO AGOTADO! (más de 15 segundos)")
                    self.vidas -= 1
                elif respuesta == solucion:
                    self.puntos += (nivel * 10)
                    print(f"✅ ¡CORRECTO! +{nivel*10} puntos")
                    print(f"   (Tiempo: {tiempo:.1f}s)")
                    nivel += 1
                    if nivel > 3:
                        print(" ¡COMPLETASTE TODOS LOS NIVELES!")
                else:
                    print(f"❌ INCORRECTO. La solución era x = {solucion}")
                    self.vidas -= 1
            except ValueError:
                print("⚠️ Ingresa un número válido.")
                self.vidas -= 1
            
            time.sleep(1.5)
        
        self.mostrar_resultados()

        print("RESULTADOS MATE-ESCAPE")
```

## `backend/app.py` — API + SQLite + Git-Sync + Servidor del OS

```python
"""BELENTANI AI — Backend Edu-OS. FastAPI + SQLite + Git content sync."""
import json, sqlite3, subprocess, pathlib
from fastapi import FastAPI
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel

ROOT = pathlib.Path(__file__).parent.parent
DB = ROOT/"backend"/"belentani.db"
app = FastAPI(title="Belentani AI Edu-OS")

def db(): 
    c = sqlite3.connect(DB); c.row_factory = sqlite3.Row
    c.executescript((ROOT/"backend"/"schema.sql").read_text(encoding="utf-8"))
    return c

def git_commit():
    try: return subprocess.run(["git","rev-parse","--short","HEAD"],cwd=ROOT,
                               capture_output=True,text=True).stdout.strip()
    except Exception: return "local"

class ScoreIn(BaseModel):
    player: int; game: str; score: int
    accuracy: float = 1.0; streak: int = 0; skills: list = []

@app.post("/players")
def create_player(name: str):
    c = db(); c.execute("INSERT OR IGNORE INTO players(name) VALUES(?)",(name,)); c.commit()
    return {"id": c.execute("SELECT id FROM players WHERE name=?",(name,)).fetchone()["id"]}

@app.post("/scores")
```

## III. IMAGE-TO-MOTION — `backend/image_to_motion.py`
Convierte el retrato estático de Belentani en **movimiento de luz líquida** (respiración de zoom, pulso de brillo carmesí, relámpagos) → GIF + spritesheet para los juegos. Sin GPU, determinista.

```python
"""BELENTANI IMAGE-TO-MOTION: retrato → GIF + spritesheet con luz líquida."""
import math, numpy as np
from PIL import Image, ImageFilter, ImageEnhance, ImageDraw

SRC, OUT, FRAMES, SIZE = "assets/belentani.png", "assets/sprites/", 24, (512, 512)
img = Image.open(SRC).convert("RGB")
a = np.asarray(img).astype(np.float32)
redness = np.clip(a[..., 0] - (a[..., 1] + a[..., 2]) / 2, 0, 255)          # máscara de energía carmesí
mask = Image.fromarray((redness / max(redness.max(), 1) * 255).astype("uint8")).filter(ImageFilter.GaussianBlur(18))

frames = []
for f in range(FRAMES):
    ph = f / FRAMES * 2 * math.pi
    z = 1.04 + 0.03 * math.sin(ph)                                          # respiración líquida (Ken Burns)
    w, h = img.size; cw, ch = int(w / z), int(h / z)
    ox = int((w - cw) / 2 + 10 * math.sin(ph)); oy = int((h - ch) / 2 + 6 * math.cos(ph * 1.3))
    base = img.crop((ox, oy, ox + cw, oy + ch)).resize(SIZE)
    m = mask.crop((ox, oy, ox + cw, oy + ch)).resize(SIZE)
    glow = ImageEnhance.Brightness(m).enhance(1.0 + 0.5 * math.sin(ph * 2)) # pulso de luz líquida
    tint = Image.composite(Image.new("RGB", SIZE, (255, 40, 70)), base, glow)
    frame = Image.blend(base, tint, 0.35 + 0.15 * math.sin(ph))
    if f % 7 == 0:                                                          # relámpago de energía
        d = ImageDraw.Draw(frame, "RGBA"); x, y, pts = np.random.randint(60, 450), 0, []
        while y < 512: pts.append((x, y)); y += 40; x += np.random.randint(-30, 30)
        d.line(pts, fill=(255, 80, 90, 140), width=3)
    frames.append(frame)

frames[0].save(OUT + "belentani_motion.gif", save_all=True, append_images=frames[1:], duration=70, loop=0)
sheet = Image.new("RGB", (SIZE[0] * 6, SIZE[1] * 4))
for i, fr in enumerate(frames): sheet.paste(fr, ((i % 6) * SIZE[0], (i // 6) * SIZE[1]))
sheet.save(OUT + "belentani_spritesheet.png")
print("✅ IMAGE-TO-MOTION OK: belentani_motion.gif + belentani_spritesheet.png")
# Nota pro: para motion corporal completo, pasar el spritesheet por EbSynth / AnimateDiff / Runway Gen-3.
```

```python
    NAME='INVASORES DE SINTAXIS'

        self.px=min(W-40,max(40,self.px+mv*380*dt))
```

```python
"""BELENTANI IMAGE-2-MOTION: hoja de referencia (4 vistas) -> turntable GIF + ciclo de caminado."""
import math, os
from PIL import Image, ImageEnhance

SRC=os.path.join(os.path.dirname(__file__),'assets','belentani_sheet.png')
OUT=os.path.join(os.path.dirname(__file__),'assets','sprites')

def crop_views(img):
    w,h=img.size
    boxes={'close':(0.0,0.0,0.38,1.0),'front':(0.38,0.0,0.63,1.0),
           'side':(0.63,0.0,0.81,1.0),'back':(0.81,0.0,1.0,1.0)}
    return {k:img.crop((int(a*w),int(b*h),int(c*w),int(d*h))) for k,(a,b,c,d) in boxes.items()}

def turntable(views, frames_per=8, size=(360,480)):
    seq=['front','side','back','side']
    vs={k:v.resize(size) for k,v in views.items()}
    frames=[]
    for i,name in enumerate(seq):
        a=vs[name]; b=vs[seq[(i+1)%len(seq)]]
        for f in range(frames_per):
            al=f/frames_per; e=al*al*(3-2*al)
            fr=Image.blend(a,b,e)
            fr=ImageEnhance.Brightness(fr).enhance(1.0+0.06*math.sin(al*math.pi))
            frames.append(fr)
    return frames

def walk_cycle(frames, bob=10, n=12):
    out=[]
    for i in range(n):
        ph=i/n*2*math.pi
        fr=frames[int(i/n*len(frames))%len(frames)]
        canvas=Image.new('RGB',fr.size,(22,1,4))
        canvas.paste(fr,(int(4*math.sin(ph)),int(bob*abs(math.sin(ph)))))
        out.append(canvas)
    return out

if __name__=='__main__':
    os.makedirs(OUT,exist_ok=True)
    img=Image.open(SRC).convert('RGB')
    tt=turntable(crop_views(img))
    wk=walk_cycle(tt)
    tt[0].save(os.path.join(OUT,'belentani_turntable.gif'),save_all=True,append_images=tt[1:],duration=60,loop=0)
    sheet=Image.new('RGB',(wk[0].width*6,wk[0].height*2))
```

```python
"""2048 MATES: fusiona tiles hasta crear el tile-objetivo de la ecuacion."""
import pygame, random
from belentani_engine import *

N=4; TS=105; OX=(W-N*TS)//2; OY=95
class Mates2048(BaseGame):
    NAME='2048 MATES'
    def setup(self):
        self.g=[[0]*N for _ in range(N)]
        self.add(); self.add(); self.new_eq()
    def add(self):
        empty=[(r,c) for r in range(N) for c in range(N) if not self.g[r][c]]
        if empty:
            r,c=random.choice(empty); self.g[r][c]=random.choice([2,2,2,4])
    def new_eq(self):
        self.target=random.choice([4,8,16,32])
        a=random.randint(1,self.target-1)
        self.eq=f"{a} + {self.target-a} = ?"
    def on_event(self,e):
        if e.type!=pygame.KEYDOWN: return
        d={pygame.K_LEFT:'L',pygame.K_RIGHT:'R',pygame.K_UP:'U',pygame.K_DOWN:'D'}.get(e.key)
        if d: self.move(d)
    def line(self,row):
        xs=[x for x in row if x]; out=[]; i=0
        while i<len(xs):
            if i+1<len(xs) and xs[i]==xs[i+1]:
                v=xs[i]*2; out.append(v); self.score+=v; self.hits+=1; self.total+=1
                if v==self.target:
                    self.add_streak(); self.score+=300; say(str(v)); self.new_eq()
                else: self.streak=0
                i+=2
            else: out.append(xs[i]); i+=1
        return out+[0]*(N-len(out))
    def move(self,d):
        old=[r[:] for r in self.g]
        if d=='L': self.g=[self.line(r) for r in self.g]
        if d=='R': self.g=[self.line(r[::-1])[::-1] for r in self.g]
        if d in ('U','D'):
            self.g=[list(x) for x in zip(*self.g)]
            if d=='U': self.g=[self.line(r) for r in self.g]
            else: self.g=[self.line(r[::-1])[::-1] for r in self.g]
            self.g=[list(x) for x in zip(*self.g)]
        if self.g!=old:
            self.add()
            if not self.can_move(): self.running=False
    def can_move(self):
        for r in range(N):
            for c in range(N):
                if not self.g[r][c]: return True
                if c+1<N and self.g[r][c]==self.g[r][c+1]: return True
                if r+1<N and self.g[r][c]==self.g[r+1][c]: return True
        return False
    def draw(self,t):

        self.screen.blit(glass(W-20,50,24,40),(10,8))
```

```python
        if opts:

            grid[nr][nc]=0; grid[wr][wc]=0; stack.append((nc,nr))
        else: stack.pop()
    return grid
```
